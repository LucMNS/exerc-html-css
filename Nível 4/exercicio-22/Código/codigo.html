      <p class="mensagem" id="mensagem">A mensagem aparecerá aqui.</p>

      <div class="botoes">
        <button type="button" id="botao-clique">Clique aqui</button>
        <button type="button" id="botao-texto">Mudar texto novamente</button>
      </div>
    </main>

    <script>
      const mensagem = document.querySelector("#mensagem");
      const botaoClique = document.querySelector("#botao-clique");
      const botaoTexto = document.querySelector("#botao-texto");

      // O JavaScript reage ao clique e altera elementos da página.
      botaoClique.addEventListener("click", () => {
        mensagem.textContent = "Você clicou no botão! A página mudou.";
        document.body.style.backgroundColor = "#d8e9ff";
      });

      botaoTexto.addEventListener("click", () => {
        mensagem.textContent = "O segundo botão alterou o texto novamente!";
      });
    </script>
