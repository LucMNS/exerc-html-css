Código do Exercício 01: index.html


<!doctype html>
<!-- O atributo lang informa o idioma principal de todo o documento. -->
<html lang="pt-br">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>Atributos HTML</title>
  </head>
  <body>
    <h1>Trabalhando com atributos HTML</h1>

    <p>
      Atributos fornecem informações extras e modificam o comportamento dos
      elementos HTML.
    </p>

    <!-- href define o endereço para o qual o link direciona. -->
    <!-- title mostra uma dica adicional quando o mouse passa sobre o link. -->
    <p>
      <a
        href="https://developer.mozilla.org/pt-BR/docs/Web/HTML"
        title="Consultar a documentação de HTML na MDN"
      >
        Consulte a documentação de HTML
      </a>
    </p>

    <!-- src define o endereço da imagem. -->
    <!-- alt descreve a imagem caso ela não carregue e auxilia leitores de tela. -->
    <!-- width define a largura da imagem em pixels. -->
    <!-- height define a altura da imagem em pixels. -->
    <img
      src="https://placehold.co/600x300/png?text=Tecnologia"
      alt="Ilustração com a palavra Tecnologia"
      title="Imagem demonstrativa de tecnologia"
      width="600"
      height="300"
    />

    <!-- lang também pode indicar o idioma específico de um trecho. -->
    <p lang="en">Technology helps people connect and learn.</p>
  </body>
</html>
