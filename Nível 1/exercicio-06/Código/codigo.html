Código do Exercício 01: index.html


<!doctype html>
<html lang="pt-br">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>Comentários em HTML</title>
  </head>
  <body>
    <!-- Início do cabeçalho da página. -->
    <header>
      <h1>Organização com comentários HTML</h1>

      <!-- A navegação fica no cabeçalho para facilitar o acesso às seções. -->
      <nav>
        <a href="#conteudo">Conteúdo</a>
        <a href="#rodape">Rodapé</a>
      </nav>
    </header>

    <!-- Início do conteúdo principal da página. -->
    <main id="conteudo">
      <!-- Esta decisão mantém o conteúdo dividido em seções semânticas. -->
      <section>
        <h2>Por que usar comentários?</h2>
        <p>
          Comentários ajudam a documentar o código e facilitam sua manutenção.
          Eles não aparecem no conteúdo visual da página.
        </p>
      </section>

      <!-- Este aviso foi comentado temporariamente e não aparece no navegador.
			<p>Este elemento será exibido futuramente.</p>
			-->
    </main>

    <!-- Início do rodapé da página. -->
    <footer id="rodape">
      <!-- O rodapé apresenta a informação de autoria. -->
      <p>Exercício de HTML Comments - PROA</p>
    </footer>
  </body>
</html>
